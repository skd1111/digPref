## 总体测试策略

### 单元测试套件

```python
# services/agent/tests/test_fiscal_compliant_scenarios.py

import pytest
from agent.audit_expert import submit_audit_task

SCENARIOS = [
    ("S1", 96, 100, "approved", False),
    ("S2", 96, 100, "approved", False),
    ("S3", 92, 100, "approved", False),
    ("S4", 91, 100, "approved", False),
    ("S5", 94, 100, "approved", False),
]


@pytest.mark.parametrize("scenario_id,min_score,max_score,expected_status,expects_hitl", SCENARIOS)
def test_compliant_scenario(scenario_id, min_score, max_score, expected_status, expects_hitl):
    """验证完全合规的业务场景走绿色快通道"""
    task = load_fixture(f"fiscal-tax/test-scenarios/合规业务-正向审核.md#{scenario_id}")
    result = submit_audit_task(task)

    assert result["status"] == expected_status
    assert min_score <= result["score"] <= max_score
    assert result["needs_hitl"] == expects_hitl
    assert len(result["compliance_violations"]) == 0
    assert result["audit_log"]["rsa_signature_v1"] is not None
    assert result["audit_log"]["sha256_evidence_hash"] is not None
    assert result["sla"] is not None  # 必须有时延记录
```
